cmake_minimum_required(VERSION 3.21)
project(ToolscreenRepack LANGUAGES C CXX RC)
include(CTest)
if(NOT DEFINED EASYINJECT_SOURCE_DIR OR NOT DEFINED INSTALLER_VERSION)
    message(FATAL_ERROR "Set EASYINJECT_SOURCE_DIR and INSTALLER_VERSION")
endif()
if(NOT INSTALLER_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "INSTALLER_VERSION must have three numeric components")
endif()
set(EASYINJECT_X64_ONLY ON CACHE BOOL "" FORCE)
set(EASYINJECT_BUILD_REDUCED_AV_HEURISTICS OFF CACHE BOOL "" FORCE)
set(EASYINJECT_ENABLE_PACKAGING_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${EASYINJECT_SOURCE_DIR}/exe" easyinject)
string(REPLACE "." "," INSTALLER_VERSION_COMMAS "${INSTALLER_VERSION}")
configure_file(installer-version.rc.in installer-version.rc @ONLY)
target_sources(EasyInjectExe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/installer-version.rc")
